def guard_contract(src):
    return all(x in src for x in (
        '/proc/sleep_graph_export_enabled()\n    return FALSE',
        'proc/owns_atom(atom/movable/A, mob/living/human/root)\n        return FALSE',
        'proc/owns_datum(datum/D, mob/living/human/root)\n        return FALSE',
        'proc/owns_list(list/L, mob/living/human/root)\n        return FALSE',
        'proc/verify_snapshot(mob/living/human/root, round_id, account, nonce)\n        return FALSE',
        '!authority.owns_atom(A, root)', '!authority.owns_datum(D, root)',
        '!authority.owns_list(L, root)', 'if(!A.verify_snapshot(H, round_id, account, nonce))',
        'if(depth > 64)', 'if(++edges > 4096)', 'entry_pins.len >= 512',
        'list_pins.len >= 256', '"state" = "unverified"')) and src.count('if(!A.verify_snapshot(H, round_id, account, nonce))') == 2

class Entry:
    def __init__(self,name):self.name=name;self.refs=[]
class GraphModel:
    """Independent ID/closure model; JSON is NOT the DM/savefile payload."""
    def __init__(self,allowed,limit=32):self.allowed=set(allowed);self.limit=limit;self.ids={};self.nodes=[]
    def encode(self,obj):
        if obj not in self.allowed: raise ValueError('external')
        if obj in self.ids:return self.ids[obj]
        if len(self.nodes)>=self.limit:raise ValueError('limit')
        number=len(self.nodes)+1;self.ids[obj]=number;self.nodes.append(None)
        self.nodes[number-1]={'name':obj.name,'refs':[self.encode(x) for x in obj.refs]}
        return number
    def payload(self,root):
        number=self.encode(root)
        body=json.dumps({'root':number,'nodes':self.nodes},sort_keys=True,separators=(',',':')).encode()
        return body,{'length':len(body),'sha256':hashlib.sha256(body).hexdigest()}

    def test_reference_boundaries_in_independent_envelope_model(self):
        def admissible(record, entry_count, list_count):
            if type(record) is not dict or set(record) not in ({'kind','ref'},{'kind','value'}):return False
            if record['kind']=='value':return set(record)=={'kind','value'} and (record['value'] is None or type(record['value']) in (int,str))
            if record['kind'] not in ('entry','list') or set(record)!={'kind','ref'}:return False
            limit=entry_count if record['kind']=='entry' else list_count
            return type(record['ref']) is int and 1<=record['ref']<=limit
        self.assertTrue(admissible({'kind':'entry','ref':2},4,2))
        self.assertTrue(admissible({'kind':'list','ref':2},4,2))
        self.assertTrue(admissible({'kind':'value','value':None},4,2))
        for value in ({'kind':'entry','ref':5},{'kind':'list','ref':3},
                      {'kind':'entry','ref':0},{'kind':'entry','ref':1.5},
                      {'kind':'turf','ref':1},{'kind':'entry','ref':1,'extra':1}):
            with self.subTest(value=value):self.assertFalse(admissible(value,4,2))

    def test_model_rejects_external_and_limit(self):
        human,item,outside=Entry('human'),Entry('item'),Entry('outside');human.refs=[item];item.refs=[outside]
        with self.assertRaises(ValueError):GraphModel([human,item]).payload(human)
        with self.assertRaises(ValueError):GraphModel([human,item,outside],limit=2).payload(human)
    def test_model_reagent_graph_preserves_alias_and_rejects_unowned_nested_members(self):
        human,item,holder,members,member,node=(Entry(name) for name in
            ('human','item','holder','member-list','member','node'))
        human.refs=[item,node]
        item.refs=[holder]
        holder.refs=[item,members]
        members.refs=[member]
        member.refs=[holder]
        node.refs=[members]
        allowed=[human,item,holder,members,member,node]
        model=GraphModel(allowed)
        body,_=model.payload(human)
        rows=json.loads(body)['nodes']
        ids={row['name']: index+1 for index,row in enumerate(rows)}
        self.assertEqual(rows[ids['holder']-1]['refs'],[ids['item'],ids['member-list']])
        self.assertEqual(rows[ids['member']-1]['refs'],[ids['holder']])
        self.assertEqual(rows[ids['node']-1]['refs'],[ids['member-list']])
        for blocked in (members,member):
            with self.subTest(blocked=blocked.name), self.assertRaises(ValueError):
                GraphModel([entry for entry in allowed if entry is not blocked]).payload(human)
